// ─────────────────────────────────────────────────────────────────────────────
// CopyLinkTutorial.tsx
//
// A small looping "show, don't tell" tutorial that teaches users how to copy
// a reply link on X. Sits in the reply-finish panel of the Engage flow.
//
// Phases (no idle, ~5s total loop):
//   0  (2000ms)  Hand slides in from off-card right → lands on Share icon
//                → click-press fires at ~1100ms → menu opens INSTANTLY
//   1  (1500ms)  Hand glides to "Copy link" menu row → click-press at ~800ms
//   2  (1500ms)  Hold "Link copied — paste it below" success state
//   → loop
//
// All styles in copy-link-tutorial.css. No external animation libraries.
// ─────────────────────────────────────────────────────────────────────────────

import * as React from "react";
import { Info, Link2, MessageCircle, ArrowUp } from "lucide-react";

import { cn } from "@/lib/cn";

import "./copy-link-tutorial.css";

export interface CopyLinkTutorialProps {
    /** Reply text shown inside the mock tweet card. Truncated to 2 lines. */
    snippet?: string;
    /** @-handle (with or without the leading "@"). Default "@yinye". */
    handle?: string;
    /** Optional className passed to the outer wrapper. */
    className?: string;
}

const PHASE_MS = [2000, 1500, 1500] as const;

export function CopyLinkTutorial({
    snippet,
    handle = "@yinye",
    className,
}: CopyLinkTutorialProps) {
    const [phase, setPhase] = React.useState<0 | 1 | 2>(0);
    const [shareClicked, setShareClicked] = React.useState(false); // also = menu open
    const [copyClicked, setCopyClicked] = React.useState(false);
    const [entered, setEntered] = React.useState(false);

    const stageRef = React.useRef<HTMLDivElement | null>(null);
    const shareRef = React.useRef<HTMLDivElement | null>(null);
    const copyLinkRef = React.useRef<HTMLDivElement | null>(null);

    const [cursorXY, setCursorXY] = React.useState({ x: 9999, y: 9999 });

    const placeOn = React.useCallback(
        (ref: React.RefObject<HTMLElement | null>) => {
            const stage = stageRef.current;
            const target = ref.current;
            if (!stage || !target) return;
            const s = stage.getBoundingClientRect();
            const t = target.getBoundingClientRect();
            setCursorXY({
                x: t.left - s.left + t.width / 2,
                y: t.top - s.top + t.height / 2,
            });
        },
        [],
    );

    // Phase loop with per-phase duration
    React.useEffect(() => {
        const t = window.setTimeout(
            () => setPhase((p) => ((p + 1) % 3) as 0 | 1 | 2),
            PHASE_MS[phase],
        );
        return () => window.clearTimeout(t);
    }, [phase]);

    // Per-phase choreography
    React.useEffect(() => {
        const timers: number[] = [];

        if (phase === 0) {
            // Loop restart: snap cursor off-card right with NO transition
            setShareClicked(false);
            setCopyClicked(false);
            setEntered(false);
            const stage = stageRef.current;
            if (stage) {
                const s = stage.getBoundingClientRect();
                setCursorXY({ x: s.width + 30, y: s.height - 18 });
            }
            // Next frame: enter + glide onto Share icon
            timers.push(
                window.setTimeout(() => {
                    setEntered(true);
                    placeOn(shareRef);
                }, 40),
            );
            // Click-pulse Share → menu opens
            timers.push(window.setTimeout(() => setShareClicked(true), 1100));
        }

        if (phase === 1) {
            timers.push(window.setTimeout(() => placeOn(copyLinkRef), 30));
            timers.push(window.setTimeout(() => setCopyClicked(true), 800));
        }

        return () => timers.forEach(window.clearTimeout);
    }, [phase, placeOn]);

    // Re-pin cursor on window resize so it stays on its current target
    React.useEffect(() => {
        const onResize = () => {
            if (!entered) return;
            if (phase === 0) placeOn(shareRef);
            else placeOn(copyLinkRef);
        };
        window.addEventListener("resize", onResize);
        return () => window.removeEventListener("resize", onResize);
    }, [phase, entered, placeOn]);

    const status = !shareClicked
        ? { txt: "Click the Share icon under your reply", emoji: "👇" }
        : !copyClicked
            ? { txt: "Pick Copy link from the menu", emoji: "🔗" }
            : { txt: "Link copied — paste it below", emoji: "✅" };

    const initial =
        (handle || "@y").replace("@", "").charAt(0).toUpperCase() || "Y";
    const preview = (
        snippet ||
        "About: SuperHi is an online learning platform that provides code, design, and project management courses for creative professionals."
    )
        .replace(/\s+/g, " ")
        .trim();

    return (
        <div className={cn("clt-wrap", className)}>
            <div ref={stageRef} className="clt-stage">
                {/* Mock tweet card */}
                <div className="clt-card">
                    <div className="clt-head">
                        <div className="clt-avatar">{initial}</div>
                        <div className="clt-handle-row">
                            <span className="clt-handle">{handle}</span>
                            <span className="clt-time">Now</span>
                        </div>
                        <span className="clt-dots">···</span>
                    </div>

                    <div className="clt-reply">{preview}</div>

                    <div className="clt-actions">
                        <span className="clt-action"><XReply /></span>
                        <span className="clt-action"><XRepost /></span>
                        <span className="clt-action"><XHeart /></span>
                        <span className="clt-action"><XView /></span>
                        <span className="clt-action"><XBookmark /></span>

                        <div
                            ref={shareRef}
                            key={`share-${shareClicked}`}
                            className={cn(
                                "clt-share",
                                shareClicked && "is-active",
                                shareClicked && "is-clicked",
                            )}
                        >
                            <XShare />
                        </div>
                    </div>

                    {/* Share-menu popover — opens INSTANTLY at click */}
                    <div className={cn("clt-menu", shareClicked && "is-open")}>
                        <MenuRow
                            icon={<MessageCircle />}
                            label="Send via Chat"
                        />
                        <MenuRow
                            ref={copyLinkRef}
                            icon={<Link2 />}
                            label="Copy link"
                            highlighted={copyClicked}
                            clicked={copyClicked}
                        />
                        <MenuRow icon={<ArrowUp />} label="Share post via…" />
                    </div>
                </div>

                {/* Hand cursor */}
                <div
                    className={cn("clt-cursor", entered && "is-entered")}
                    style={{ left: cursorXY.x, top: cursorXY.y }}
                >
                    <span
                        key={`cursor-press-${shareClicked}-${copyClicked}`}
                        className={cn(
                            "clt-cursor-press",
                            (shareClicked || copyClicked) && "is-clicked",
                        )}
                    >
                        <HandCursor />
                    </span>
                </div>

                {/* Progress dots */}
                <div className="clt-progress">
                    {[0, 1, 2].map((i) => (
                        <span
                            key={i}
                            className={cn(
                                "clt-progress-dot",
                                i <= phase && "is-on",
                            )}
                        />
                    ))}
                </div>

                {/* Status line */}
                <div className="clt-status">
                    <Info className="clt-status-icon" />
                    <span
                        key={status.txt}
                        className={cn(
                            "clt-status-text",
                            copyClicked && "is-done",
                        )}
                    >
                        <span aria-hidden className="clt-status-emoji">
                            {status.emoji}
                        </span>
                        <span>{status.txt}</span>
                    </span>
                </div>
            </div>
        </div>
    );
}

// ── Sub-components ──────────────────────────────────────────────────────────
interface MenuRowProps {
    icon: React.ReactNode;
    label: string;
    highlighted?: boolean;
    clicked?: boolean;
}

const MenuRow = React.forwardRef<HTMLDivElement, MenuRowProps>(
    function MenuRow({ icon, label, highlighted, clicked }, ref) {
        return (
            <div
                ref={ref}
                key={`mr-${highlighted}-${clicked}`}
                className={cn(
                    "clt-menu-row",
                    highlighted && "is-highlighted",
                    clicked && "is-clicked",
                )}
            >
                <span className="clt-menu-icon">{icon}</span>
                <span className="clt-menu-label">{label}</span>
            </div>
        );
    },
);
