/* Pivot at the top-center hanging point (12,3). */ .bm-shape { transform-box: view-box; transform-origin: 12px 3px; } /* Playful pendulum: top fixed, the notch swings L↔R, decaying. */ @keyframes bm-wag { 0% { transform: rotate(0deg); } 8% { transform: rotate(0deg); } /* anticipation */ 22% { transform: rotate(11deg); } /* swing right */ 40% { transform: rotate(-8deg); } /* swing left */ 56% { transform: rotate(5deg); } 70% { transform: rotate(-3deg); } 82% { transform: rotate(1.5deg); } 100% { transform: rotate(0deg); } /* settle + rest */ } .bm:hover .bm-shape { animation: bm-wag 1600ms cubic-bezier(.36,.07,.19,.97) infinite; }
