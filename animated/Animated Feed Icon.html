/* CSS — each bar pulses scaleY around its own midpoint y=8 */ .feed svg .b-outer-l { transform-origin: 3px 8px; } .feed svg .b-mid-l { transform-origin: 5.5px 8px; } .feed svg .b-center { transform-origin: 8px 8px; } .feed svg .b-mid-r { transform-origin: 10.5px 8px; } .feed svg .b-outer-r { transform-origin: 13px 8px; } .feed:hover svg .feed-bar { animation: feed-pulse 1.4s linear infinite; } /* wave travels INWARD: outer first, center last */ .feed:hover svg .b-outer-l, .b-outer-r { animation-delay: 0ms; } .feed:hover svg .b-mid-l, .b-mid-r { animation-delay: 140ms; } .feed:hover svg .b-center { animation-delay: 280ms; } /* Sine-shaped scaleY → continuous velocity, no hitches */ @keyframes feed-pulse { 0% { transform: scaleY(1); } 12.5% { transform: scaleY(1.17); } 25% { transform: scaleY(1.22); } /* tall peak */ 37.5% { transform: scaleY(1.13); } 50% { transform: scaleY(1); } /* through rest */ 62.5% { transform: scaleY(0.78); } 75% { transform: scaleY(0.6); } /* short trough */ 87.5% { transform: scaleY(0.78); } 100% { transform: scaleY(1); } }
