/* Exhaust flame — elastic burst + contraction, rooted at the nozzle */ .r-flame { transform-box: fill-box; transform-origin: 50% 0%; } @keyframes rocket-flame { 0% { transform: scaleY(0.45); } 22% { transform: scaleY(1.28); } /* fast burst, overshoot */ 40% { transform: scaleY(0.82); } /* spring back */ 58% { transform: scaleY(1.12); } /* bounce */ 74% { transform: scaleY(0.94); } 88% { transform: scaleY(1.02); } 100% { transform: scaleY(0.45); } /* elastic contraction */ } /* spring easing on the whole loop */ .rocket:hover .r-flame { animation: rocket-flame 850ms cubic-bezier(.34,1.56,.64,1) infinite; } /* Porthole flicker */ @keyframes rocket-window { 0%,100%{opacity:1} 50%{opacity:.2} } /* Speed lines: translate backward + stretch their own length (dasharray) */ @keyframes rocket-speed { 0% { opacity:0; transform:translate(40px,-40px); stroke-dasharray:4 40; } /* appears at nose */ 50% { stroke-dasharray:34 40; } /* stretched mid-sweep */ 100% { opacity:0; transform:translate(-44px,44px); stroke-dasharray:5 40; } /* vanishes by flame */ }
