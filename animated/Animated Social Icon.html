/* CSS */ .social svg .social-avatar, .social svg .social-bar-a, .social svg .social-bar-b, .social svg .social-bar-c { transform-box: fill-box; transform-origin: left center; } .social svg .social-avatar { transform-origin: center center; } .social:hover svg .social-avatar { animation: social-avatar-pop 1.6s ease-in-out infinite; } .social:hover svg .social-bar-a { animation: social-bar-type 1.6s ease-in-out 0ms infinite; } .social:hover svg .social-bar-b { animation: social-bar-type 1.6s ease-in-out 120ms infinite; } .social:hover svg .social-bar-c { animation: social-bar-type 1.6s ease-in-out 240ms infinite; } @keyframes social-avatar-pop { 0% { transform: scale(1); } 18% { transform: scale(1.08); } 35% { transform: scale(1); } 100% { transform: scale(1); } } @keyframes social-bar-type { 0% { transform: scaleX(1); } /* visible */ 15% { transform: scaleX(1); } /* hold */ 22% { transform: scaleX(0); } /* erase */ 60% { transform: scaleX(1); } /* type back */ 100% { transform: scaleX(1); } }
